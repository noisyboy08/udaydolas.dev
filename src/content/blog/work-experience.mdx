## Overview

A clean, collapsible work experience component. Each company can have multiple roles (positions), each with a type icon, employment period, markdown description, and skill tags.

## Usage

```tsx
import { WorkExperience } from "@/registry/work-experience";

const experiences = [
  {
    id: "company-a",
    companyName: "Company A",
    companyLogo: "https://example.com/logo.svg",
    positions: [
      {
        id: "pos-1",
        title: "Software Engineer",
        employmentPeriod: "01.2024 — present",
        employmentType: "Full-time",
        icon: "code",
        description: "- Built core product features\n- Led frontend architecture",
        skills: ["React", "TypeScript", "Next.js"],
        isExpanded: true,
      },
    ],
    isCurrentEmployer: true,
  },
];

export default function Demo() {
  return <WorkExperience experiences={experiences} />;
}
```
